cmake_minimum_required(VERSION 3.22)
project(rnbo_esp32s3_host LANGUAGES C CXX)
set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
include(CTest)
option(RNBO_S3_SANITIZERS "Host AddressSanitizer/UBSan" OFF)
if(RNBO_S3_SANITIZERS AND CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
  add_link_options(-fsanitize=address,undefined)
endif()
add_executable(phase_trunc_test tests/native/phase_trunc_test.cpp)
target_include_directories(phase_trunc_test PRIVATE runtime)
target_compile_options(phase_trunc_test PRIVATE -UNDEBUG -fno-fast-math)
add_test(NAME phase_trunc COMMAND phase_trunc_test)
add_library(pcm_output STATIC firmware/components/pcm_output/pcm_output.c)
target_include_directories(pcm_output PUBLIC firmware/components/pcm_output/include)
target_compile_options(pcm_output PRIVATE -Wall -Wextra -Werror -fno-fast-math)
if(UNIX AND NOT APPLE)
  target_link_libraries(pcm_output PUBLIC m)
endif()
add_executable(pcm_output_test tests/native/pcm_output_test.c)
target_compile_options(pcm_output_test PRIVATE -UNDEBUG)
target_link_libraries(pcm_output_test PRIVATE pcm_output)
add_test(NAME pcm_output COMMAND pcm_output_test)
add_library(pcm_capture STATIC runtime/pcm_capture.c)
target_include_directories(pcm_capture PUBLIC runtime)
target_compile_options(pcm_capture PRIVATE -Wall -Wextra -Werror)
add_executable(pcm_capture_test tests/native/capture_test.c)
target_compile_options(pcm_capture_test PRIVATE -UNDEBUG)
target_link_libraries(pcm_capture_test PRIVATE pcm_capture)
add_test(NAME pcm_capture COMMAND pcm_capture_test)
add_library(audio_lifecycle STATIC firmware/components/audio_lifecycle/audio_lifecycle.c)
target_include_directories(audio_lifecycle PUBLIC firmware/components/audio_lifecycle/include)
target_compile_options(audio_lifecycle PRIVATE -Wall -Wextra -Werror)
add_executable(audio_lifecycle_test tests/native/lifecycle_test.c)
target_compile_options(audio_lifecycle_test PRIVATE -UNDEBUG)
target_link_libraries(audio_lifecycle_test PRIVATE audio_lifecycle)
add_test(NAME audio_lifecycle COMMAND audio_lifecycle_test)

set(RNBO_S3_EXPORT_DIR "" CACHE PATH "Validated prepared RNBO Minimal C++ export")
set(RNBO_S3_MIDI_TEST_VOICES "0" CACHE STRING "MIDI envelope fixture voices (0 disables fixture-specific tests)")
if(RNBO_S3_EXPORT_DIR)
  get_filename_component(RNBO_S3_EXPORT_DIR "${RNBO_S3_EXPORT_DIR}" ABSOLUTE)
  foreach(required patch_config.h rnbo_source.h common/RNBO_Common.h)
    if(NOT EXISTS "${RNBO_S3_EXPORT_DIR}/${required}")
      message(FATAL_ERROR "Missing prepared RNBO export file: ${required}")
    endif()
  endforeach()
  add_library(rnbo_static STATIC module-sdk/wrapper/rnbo_s3_static.cpp)
  target_include_directories(rnbo_static PUBLIC module-sdk/include PRIVATE "${RNBO_S3_EXPORT_DIR}" "${RNBO_S3_EXPORT_DIR}/common")
  target_compile_options(rnbo_static PRIVATE -fno-fast-math -ffp-contract=off -fno-strict-aliasing -fno-exceptions -fno-rtti)
  add_executable(rnbo_render tests/native/render.cpp)
  target_link_libraries(rnbo_render PRIVATE rnbo_static pcm_output)
  add_executable(rnbo_contract_test tests/native/contract_test.cpp)
  target_compile_options(rnbo_contract_test PRIVATE -UNDEBUG)
  target_link_libraries(rnbo_contract_test PRIVATE rnbo_static)
  add_test(NAME rnbo_contract COMMAND rnbo_contract_test)
  if(RNBO_S3_MIDI_TEST_VOICES MATCHES "^(1|2|4)$")
    add_executable(rnbo_midi_test tests/native/midi_test.cpp)
    target_link_libraries(rnbo_midi_test PRIVATE rnbo_static)
    target_compile_options(rnbo_midi_test PRIVATE -UNDEBUG)
    add_test(NAME rnbo_midi COMMAND rnbo_midi_test ${RNBO_S3_MIDI_TEST_VOICES})
  elseif(NOT RNBO_S3_MIDI_TEST_VOICES STREQUAL "0")
    message(FATAL_ERROR "RNBO_S3_MIDI_TEST_VOICES must be 0, 1, 2 or 4")
  endif()
  file(READ "${RNBO_S3_EXPORT_DIR}/manifest.json" export_manifest)
  string(FIND "${export_manifest}" "rnbo-1.4.4-list-cross-capacity-assignment-v1" fixed_runtime)
  if(fixed_runtime GREATER_EQUAL 0)
    add_executable(rnbo_runtime_test tests/native/runtime_test.cpp)
    target_include_directories(rnbo_runtime_test PRIVATE module-sdk/include "${RNBO_S3_EXPORT_DIR}" "${RNBO_S3_EXPORT_DIR}/common")
    target_compile_options(rnbo_runtime_test PRIVATE -UNDEBUG -fno-fast-math -ffp-contract=off -fno-strict-aliasing -fno-exceptions -fno-rtti)
    add_test(NAME rnbo_runtime COMMAND rnbo_runtime_test)
  endif()
endif()
