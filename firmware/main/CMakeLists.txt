    list(APPEND app_requires rnbo_host)
endif()

    list(APPEND app_requires player_loader player_store player_network json mbedtls)
    set(app_main_source "player_main.c")

    set(app_main_source "main.c")

    target_compile_definitions(${COMPONENT_LIB} PRIVATE RNBO_S3_PLAYER=1 RNBO_S3_WIFI=1)
endif()

    message(FATAL_ERROR "Set startup volume from 0 to 100 percent")
endif()

if(NOT "$ENV{RNBO_S3_EXPORT_DIR}" STREQUAL "")
    target_compile_definitions(${COMPONENT_LIB} PRIVATE RNBO_S3_STATIC_MODULE=1)
    file(READ "$ENV{RNBO_S3_EXPORT_DIR}/manifest.json" prepared_manifest)
    string(JSON source_identity GET "${prepared_manifest}" source_tree_sha256)
    string(LENGTH "${source_identity}" identity_length)
    if(NOT source_identity MATCHES "^[0-9a-f]+$" OR NOT identity_length EQUAL 64)
        message(FATAL_ERROR "Prepared source identity must be a SHA-256 digest")
    endif()
    target_compile_definitions(${COMPONENT_LIB} PRIVATE RNBO_S3_MODULE_SOURCE_SHA256="${source_identity}")
    if(NOT RNBO_S3_ARENA_BYTES MATCHES "^[0-9]+$" OR RNBO_S3_ARENA_BYTES LESS 4096 OR RNBO_S3_ARENA_BYTES GREATER 4194304)
        message(FATAL_ERROR "Use an explicit RNBO arena budget from 4096 to 4194304 bytes")
    endif()
    target_compile_definitions(${COMPONENT_LIB} PRIVATE RNBO_S3_ARENA_BYTES=${RNBO_S3_ARENA_BYTES})
endif()
